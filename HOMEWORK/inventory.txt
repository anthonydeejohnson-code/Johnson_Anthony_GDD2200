using System;

//abstract player class from class notes 
public abstract class player{
    private int HP = 20;
    bool isPoisoned = false;
    bool isUndead = false;
    bool isStone = false;
    bool isDisease = false;

 //getter
    public int getHP(){
        return HP;
    }    
    //setter
    public void setHP(int newHP){
        HP = newHP;
    }  
    
    public bool Poisoned(){
        return isPoisoned;
    }
    
    public void setPoisoned(){
        isPoisoned = !isPoisoned;
    }
    public abstract void spcATK();

    public static void AddItem(List<Item> inventory, Item newItem)
{
    foreach(Item item in inventory)
    {
        if(item.GetType() == newItem.GetType())
        {
            item.amount++;
            return;
        }
    }

    newItem.amount = 1;
    inventory.Add(newItem);
}
}

public abstract class Item{

    public string name;
    public int amount;
    public int itemID;

    public bool isUsed;

    public abstract void ItemEffect(player play1);
}

public class Potion : Item
{
    public Potion()
    {
        name = "Potion";
    }

    public override void ItemEffect(player play1)
    {
        Console.WriteLine("Players Current HP");
        Console.WriteLine(play1.getHP());

        int hp = play1.getHP();
        hp = hp + 20;

        play1.setHP(hp);

        Console.WriteLine("Healed player");
        Console.WriteLine(play1.getHP());
    }
}

public class Antidote : Item
{
    public Antidote()
    {
        name = "Antidote";
    }

    public override void ItemEffect(player play1)
    {
        if (play1.Poisoned())
        {
            Console.WriteLine("Poisoned");

            play1.setPoisoned();

            Console.WriteLine("Poison Healed");
        }
        else
        {
            Console.WriteLine("Not Poisoned");
        }
    }
}
